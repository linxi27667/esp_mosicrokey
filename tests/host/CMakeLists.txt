cmake_minimum_required(VERSION 3.20)
project(mosicrokey_host LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_C_STANDARD 11)
get_filename_component(ROOT "../.." ABSOLUTE)
set(MANAGED "${ROOT}/managed_components")
file(GLOB LUA "${MANAGED}/georgik__lua/lua/*.c")
list(FILTER LUA EXCLUDE REGEX "/(lua|luac|onelua)\\.c$")
add_library(lua STATIC ${LUA})
target_include_directories(lua PUBLIC "${MANAGED}/georgik__lua/lua" "${MANAGED}/georgik__lua/include")
target_compile_definitions(lua PUBLIC CONFIG_LUA_MAXSTACK=4096)
add_library(cjson STATIC "${MANAGED}/espressif__cjson/cJSON/cJSON.c")
target_include_directories(cjson PUBLIC "${MANAGED}/espressif__cjson/cJSON")
add_library(game STATIC "${ROOT}/components/game_service/game_library.cc" "${ROOT}/components/game_service/game_vm.cc" "${ROOT}/components/game_service/game_diagnostics.cc" "${ROOT}/components/game_service/raster.cc")
target_include_directories(game PUBLIC "${ROOT}/components/mosicrokey_contracts/include" "${ROOT}/components/game_service/include" "${ROOT}/components/game_service")
target_link_libraries(game PUBLIC lua cjson)
add_executable(game_tests game_tests.cc)
target_link_libraries(game_tests PRIVATE game)
add_executable(snake_probe snake_probe.cc)
target_link_libraries(snake_probe PRIVATE game)
enable_testing()
add_executable(hal_tests hal_tests.cc "${ROOT}/components/board_support/hardware/hal_key.cc"
    "${ROOT}/components/board_support/hardware/hal_uart.cc")
target_include_directories(hal_tests PRIVATE mocks "${ROOT}/components/board_support/include")
add_test(NAME hal_tests COMMAND hal_tests)
add_executable(storage_hal_tests storage_hal_tests.cc "${ROOT}/components/board_support/hardware/hal_storage.cc")
target_include_directories(storage_hal_tests PRIVATE mocks "${ROOT}/components/board_support/include")
add_test(NAME storage_hal_tests COMMAND storage_hal_tests)
add_executable(display_hal_tests display_hal_tests.cc "${ROOT}/components/board_support/hardware/hal_display.cc")
target_include_directories(display_hal_tests PRIVATE mocks "${ROOT}/components/board_support/include")
add_test(NAME display_hal_tests COMMAND display_hal_tests)
add_test(NAME game_tests COMMAND game_tests "${CMAKE_BINARY_DIR}/test-card" "${ROOT}/examples/snake/main.lua")
set(CLAW_SRC "${ROOT}/third_party/esp-claw/components/claw_modules/claw_core/src")
add_executable(llm_retry_tests llm_retry_tests.c "${CLAW_SRC}/llm/backends/claw_llm_backend_openai_compatible.c")
target_include_directories(llm_retry_tests PRIVATE retry_mocks "${CLAW_SRC}")
target_link_libraries(llm_retry_tests PRIVATE cjson)
add_test(NAME llm_retry_tests COMMAND llm_retry_tests)
add_executable(llm_transport_tests llm_transport_tests.c "${CLAW_SRC}/llm/claw_llm_http_transport.c")
target_include_directories(llm_transport_tests PRIVATE transport_mocks retry_mocks "${CLAW_SRC}")
target_compile_definitions(llm_transport_tests PRIVATE CONFIG_HTTP_REUSE_ENABLE=1)
target_link_libraries(llm_transport_tests PRIVATE cjson)
add_test(NAME llm_transport_tests COMMAND llm_transport_tests)
set_tests_properties(game_tests PROPERTIES TIMEOUT 30)
add_executable(app_tests app_tests.cc "${ROOT}/components/mosicrokey_app/app.cc" "${ROOT}/components/mosicrokey_app/task/app_task.cc")
target_link_libraries(app_tests PRIVATE cjson)
target_include_directories(app_tests PRIVATE mocks "${ROOT}/components/mosicrokey_contracts/include"
    "${ROOT}/components/espclaw_compoment"
    "${ROOT}/components/mosicrokey_app" "${ROOT}/components/mosicrokey_app/include" "${ROOT}/components/board_support/include" "${ROOT}/components/mosicrokey_ui/include")
add_test(NAME app_tests COMMAND app_tests)
set_tests_properties(app_tests PROPERTIES TIMEOUT 30)
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "")
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
add_subdirectory("${MANAGED}/lvgl__lvgl" "lvgl")
add_executable(ui_capture ui_capture.cc "${ROOT}/components/mosicrokey_ui/view.cc" "${ROOT}/assets/fonts/mosicrokey_font14.c")
target_include_directories(ui_capture PRIVATE "${ROOT}/components/mosicrokey_contracts/include" "${ROOT}/components/mosicrokey_ui")
target_link_libraries(ui_capture PRIVATE game lvgl::lvgl)
add_test(NAME ui_capture COMMAND ui_capture "${CMAKE_BINARY_DIR}/screenshots" "${ROOT}/examples/snake/main.lua")
